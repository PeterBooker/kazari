{{- /* Index of the term in force at a time; before the first start of the year it is the last term. Usage: partial "sekki.html" now. Keep in step with the inline script in head.html. */ -}} {{- $md := .Format "01-02" -}} {{- $n := -1 -}}{{- $best := "" -}} {{- $lastN := 0 -}}{{- $last := "" -}} {{- range $i, $t := hugo.Data.sekki.terms -}} {{- if and (le $t.start $md) (gt $t.start $best) }}{{ $best = $t.start }}{{ $n = $i }}{{ end -}} {{- if gt $t.start $last }}{{ $last = $t.start }}{{ $lastN = $i }}{{ end -}} {{- end -}} {{- if lt $n 0 }}{{ $n = $lastN }}{{ end -}} {{- return $n -}}
